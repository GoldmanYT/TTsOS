% Задание характеристик
Fd = 2000;  % частота дискретизации

Fs = [300, 700];  % границы полосы непропускания
Rp = 0.2;  % неравномерность в полосе пропускания

a = [0 1 0];  % АЧХ: стоп, пасс, стоп
dev = [10^(-Rs/20) 10^(-Rp/20) 10^(-Rs/20)];  % допустимые отклонения (ИСПРАВЛЕНО: второй элемент для Rp)

% Проектирование фильтра

%% ==================== ПРОГРАММНАЯ РЕАЛИЗАЦИЯ ФИЛЬТРА ====================

% 2.1. Прямая форма (Direct Form)
function y = direct_form(x, b)
    persistent buf
    if isempty(buf)
        buf = zeros(1, length(b));
    end
    y = 0;
    for i = 1:length(b)
        y = y + b(i) * buf(i);
    end
    buf = [x, buf(1:end-1)];
end

% 2.2. Каноническая форма (для КИХ совпадает с прямой)
function y = canonical_form(x, b)
    persistent buf
    if isempty(buf)
        buf = zeros(1, length(b));
    end
    y = 0;
    for i = 1:length(b)
        y = y + b(i) * buf(i);
    end
    buf = [x, buf(1:end-1)];
end

% 2.3. Последовательная форма (каскад секций 2-го порядка)
function y = serial_form(x, sos, g)
    persistent bufs
    n_sections = size(sos, 1);
    if isempty(bufs)
        bufs = zeros(n_sections, 2);
    end
    w = g * x;
    for i = 1:n_sections
        b0 = sos(i,1); b1 = sos(i,2); b2 = sos(i,3);
        a1 = sos(i,5); a2 = sos(i,6);
        
        y_sec = b0 * w + b1 * bufs(i,1) + b2 * bufs(i,2) - a1 * bufs(i,1) - a2 * bufs(i,2);
        bufs(i,2) = bufs(i,1);
        bufs(i,1) = y_sec;
        w = y_sec;
    end
    y = w;
end

% 2.4. Параллельная форма
function y = parallel_form(x, Bi, Ai, k)
    persistent bufp
    n_sections = size(Bi, 1);
    if isempty(bufp)
        bufp = zeros(n_sections, 2);
    end
    y = k * x;
    for i = 1:n_sections
        b0 = Bi(i,1); b1 = Bi(i,2);
        a1 = Ai(i,2); a2 = Ai(i,3);
        
        y_sec = b0 * x + b1 * bufp(i,1) - a1 * bufp(i,1) - a2 * bufp(i,2);
        bufp(i,2) = bufp(i,1);
        bufp(i,1) = y_sec;
        y = y + y_sec;
    end
end

% Получение коэффициентов для последовательной формы
[sos, g] = tf2sos(b, 1);

% Получение коэффициентов для параллельной формы
[r, p, k_par] = residuez(b, 1);
Bi = [];
Ai = [];
i = 1;
while i <= length(r)
    if abs(imag(p(i))) > 1e-10  % комплексно-сопряженная пара
        b_sec = [real(r(i)+r(i+1)), -2*real(r(i)*conj(p(i)))];
        a_sec = [1, -2*real(p(i)), abs(p(i))^2];
        Bi = [Bi; b_sec];
        Ai = [Ai; a_sec];
        i = i + 2;
    else  % действительный полюс
        Bi = [Bi; [real(r(i)), 0]];
        Ai = [Ai; [1, -real(p(i)), 0]];
        i = i + 1;
    end
end

N_test = 100;

sin_stop_test = sin(2 * pi * 800 / Fd * (0:N_test-1));

% Инициализация буферов через вызов функций (очистка persistent)
clear direct_form canonical_form serial_form parallel_form

% Вычисление реакций для всех форм

y_stop_dir = zeros(1, N_test);
for i = 1:N_test
    y_imp_dir(i) = direct_form(impulse_test(i), b);
end
clear direct_form
for i = 1:N_test
    y_step_dir(i) = direct_form(step_test(i), b);
end
clear direct_form
for i = 1:N_test
    y_pass_dir(i) = direct_form(sin_pass_test(i), b);
end
clear direct_form
for i = 1:N_test
    y_stop_dir(i) = direct_form(sin_stop_test(i), b);
end

% Каноническая форма
y_imp_can = zeros(1, N_test);
for i = 1:N_test
    y_imp_can(i) = canonical_form(impulse_test(i), b);
end

% Последовательная форма
y_imp_ser = zeros(1, N_test);
for i = 1:N_test
    y_imp_ser(i) = serial_form(impulse_test(i), sos, g);

% Параллельная форма
y_imp_par = zeros(1, N_test);
for i = 1:N_test
    y_imp_par(i) = parallel_form(impulse_test(i), Bi, Ai, k_par);
end
